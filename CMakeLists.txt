cmake_minimum_required(VERSION 3.21)
project(cardputerzero-mesh VERSION 0.1.0 LANGUAGES C CXX)
option(MESH_DESKTOP "Build the SDL preview" OFF)
include(FetchContent)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/src/lv_conf.h" CACHE FILEPATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.5.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(lvgl)
add_executable(mesh-ui src/main.cpp src/linux_input.cpp)
target_compile_features(mesh-ui PRIVATE cxx_std_17)
target_link_libraries(mesh-ui PRIVATE lvgl::lvgl)
target_compile_definitions(mesh-ui PRIVATE USE_DESKTOP=$<BOOL:${MESH_DESKTOP}>)
target_compile_definitions(lvgl PUBLIC USE_DESKTOP=$<BOOL:${MESH_DESKTOP}>)
if(MESH_DESKTOP)
    find_package(SDL2 REQUIRED CONFIG)
    target_link_libraries(lvgl PUBLIC SDL2::SDL2)
else()
    add_library(mesh-loopback SHARED src/loopback.c)
    target_link_libraries(mesh-loopback PRIVATE ${CMAKE_DL_LIBS})
endif()
